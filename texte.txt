local function Create(className, props, children)
    local inst = Instance.new(className)
    for k, v in pairs(props or {}) do inst[k] = v end
    for _, child in ipairs(children or {}) do child.Parent = inst end
    return inst
end
local function Tween(instance, props, info)
    local t = TweenService:Create(instance, info or CONFIG.TweenInfo, props); t:Play(); return t
end
local function Corner(parent, radius)
    return Create("UICorner", { CornerRadius = UDim.new(0, radius or 12), Parent = parent })
end
local function Stroke(parent, color, thickness, transparency)
    return Create("UIStroke", { Color = color or CONFIG.Colors.Primary, Thickness = thickness or 2,
        Transparency = transparency or 0, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = parent })
end
local function Padding(parent, top, bottom, left, right)
    return Create("UIPadding", { PaddingTop = UDim.new(0, top or 8), PaddingBottom = UDim.new(0, bottom or 8),
        PaddingLeft = UDim.new(0, left or 12), PaddingRight = UDim.new(0, right or 12), Parent = parent })
end
local function MakeDraggable(frame, dragArea)
    dragArea = dragArea or frame
    local dragging, dragStart, startPos = false, nil, nil
    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

        Tween(frame, { BackgroundTransparency = 1 }); Tween(t, { TextTransparency = 1 }); Tween(m, { TextTransparency = 1 })

-- BACKGROUND PROGRAM
local function InitBackgroundProgram(parent, windowW, windowH)
    local container = Create("Frame", { Name = "BGProgram", Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1, ClipsDescendants = true, ZIndex = 1, Parent = parent })
    local auras = {}
    for i = 1, 3 do
        local aura = Create("Frame", { Size = UDim2.new(2, 0, 2, 0), Position = UDim2.new(-0.5, 0, -0.5, 0),
            BackgroundColor3 = (i == 1) and CONFIG.Colors.Primary or (i == 2) and CONFIG.Colors.Secondary or CONFIG.Colors.Success,
            BackgroundTransparency = 0.95, ZIndex = 1, Parent = container })
        local grad = Create("UIGradient", { Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1), NumberSequenceKeypoint.new(0.5, 0), NumberSequenceKeypoint.new(1, 1) }),
            Rotation = math.random(0, 360), Parent = aura })
        table.insert(auras, { frame = aura, gradient = grad, speed = math.random(8, 20) })
    end
    local dots = {}
    local dotCount = IS_MOBILE and 6 or 10
    for i = 1, dotCount do
        local dotSize = math.random(4, 8)
        local dot = Create("Frame", { Size = UDim2.new(0, dotSize, 0, dotSize),
            BackgroundColor3 = (i % 2 == 0) and CONFIG.Colors.Primary or CONFIG.Colors.Secondary,
            BackgroundTransparency = 0.15, ZIndex = 2, Parent = container })
        Corner(dot, dotSize / 2)
        table.insert(dots, { frame = dot, angle = (i / dotCount) * math.pi * 2, radius = math.random(80, 160),
            speed = (math.random() > 0.5) and 0.6 or -0.6,
            centerX = 0.5 + (math.random() - 0.5) * 0.1, centerY = 0.5 + (math.random() - 0.5) * 0.1 })
    end
    local chars = { "0","1","λ","Δ","Σ","✦","✧","+" }
    local rain = {}
    local colCount = IS_MOBILE and 6 or 12
    for i = 1, colCount do
        local label = Create("TextLabel", { Size = UDim2.new(0, 20, 0, 400),
            Position = UDim2.new(i / colCount, 0, 0, math.random(-400, 0)),
            BackgroundTransparency = 1, Text = "", TextColor3 = CONFIG.Colors.Primary,
            TextTransparency = 0.75, Font = CONFIG.Fonts.Mono, TextSize = 14,
            TextXAlignment = Enum.TextXAlignment.Center, ZIndex = 1, Parent = container })
        local content = ""
        for _ = 1, 30 do content = content .. chars[math.random(#chars)] .. "\n" end
        label.Text = content
        table.insert(rain, { label = label, speed = math.random(40, 100) })
    end
    local cornerGlows = {}
    local positions = { { x = -0.2, y = -0.2, rot = 45 }, { x = 0.8, y = -0.2, rot = 135 },
        { x = -0.2, y = 0.8, rot = 315 }, { x = 0.8, y = 0.8, rot = 225 } }
    for _, pos in ipairs(positions) do
        local glow = Create("Frame", { Size = UDim2.new(0.4, 0, 0.4, 0), Position = UDim2.new(pos.x, 0, pos.y, 0),
            BackgroundColor3 = CONFIG.Colors.Primary, BackgroundTransparency = 0.9, ZIndex = 1, Parent = container })
        Create("UIGradient", { Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0), NumberSequenceKeypoint.new(1, 1) }),
            Rotation = pos.rot, Parent = glow })
        table.insert(cornerGlows, { frame = glow, phase = math.random() * math.pi * 2 })
    end
    local elapsed = 0
    RunService.RenderStepped:Connect(function(dt)
        if not MainFrame.Visible then return end
        elapsed = elapsed + dt
        for _, data in ipairs(auras) do
            data.gradient.Rotation = (data.gradient.Rotation + data.speed * dt) % 360
            data.frame.BackgroundTransparency = math.clamp(0.94 + math.sin(elapsed * 1.5 + data.speed) * 0.04, 0.88, 0.99)
        end
        for _, data in ipairs(dots) do
            data.angle = data.angle + data.speed * dt
            data.frame.Position = UDim2.new(data.centerX + math.cos(data.angle) * (data.radius / windowW), -3,
                data.centerY + math.sin(data.angle) * (data.radius / windowH), -3)
            data.frame.BackgroundTransparency = 0.1 + math.abs(math.sin(elapsed * 2 + data.angle)) * 0.4
        end
        for _, data in ipairs(rain) do
            local p = data.label.Position
            local newY = p.Y.Offset + data.speed * dt
            if newY > windowH + 100 then newY = -400 end
            data.label.Position = UDim2.new(p.X.Scale, 0, 0, newY)
        end
        for _, data in ipairs(cornerGlows) do
            data.frame.BackgroundTransparency = 0.82 + math.sin(elapsed * 1.2 + data.phase) * 0.1
        end
        if MainStroke then MainStroke.Transparency = 0.1 + math.abs(math.sin(elapsed * 1.8)) * 0.3 end
    end)
end

    end)
    local label = Create("TextLabel", { Size = UDim2.new(1, -labelX - 4, 1, 0), Position = UDim2.new(0, labelX, 0, 0),
        BackgroundTransparency = 1, Text = name, TextColor3 = CONFIG.Colors.TextDim,

        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 4, Parent = btn })

local function CreateToggle(parent, text, default, callback)
    local frame = Create("Frame", { Size = UDim2.new(1, 0, 0, 45), BackgroundColor3 = CONFIG.Colors.Surface,
        BackgroundTransparency = 0.2, ZIndex = 3, Parent = parent })
    Corner(frame, 8)
    Create("TextLabel", { Size = UDim2.new(1, -80, 1, 0), Position = UDim2.new(0, 12, 0, 0),
        BackgroundTransparency = 1, Text = text, TextColor3 = CONFIG.Colors.Text,
        Font = CONFIG.Fonts.Body, TextSize = IS_MOBILE and 13 or 14,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 4, Parent = frame })
    local toggleBg = Create("Frame", { Size = UDim2.new(0, 44, 0, 24), Position = UDim2.new(1, -58, 0.5, -12),
        BackgroundColor3 = CONFIG.Colors.SurfaceLight, ZIndex = 4, Parent = frame })
    Corner(toggleBg, 12)
    local knob = Create("Frame", { Size = UDim2.new(0, 20, 0, 20), Position = UDim2.new(0, 2, 0.5, -10),
        BackgroundColor3 = CONFIG.Colors.TextDim, ZIndex = 5, Parent = toggleBg })
    Corner(knob, 10)
    local enabled = default or false
    local function update()
        Tween(toggleBg, { BackgroundColor3 = enabled and CONFIG.Colors.Success or CONFIG.Colors.SurfaceLight }, CONFIG.TweenInfoFast)
        Tween(knob, { Position = enabled and UDim2.new(1, -22, 0.5, -10) or UDim2.new(0, 2, 0.5, -10),
            BackgroundColor3 = enabled and CONFIG.Colors.Text or CONFIG.Colors.TextDim }, CONFIG.TweenInfoFast)
        if callback then callback(enabled) end
    end
    local clickArea = Create("TextButton", { Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1,
        Text = "", ZIndex = 6, Parent = frame })
    clickArea.MouseButton1Click:Connect(function() enabled = not enabled; update() end)
    update()
    return frame
end

local function CreateSlider(parent, text, min, max, default, callback)
    local frame = Create("Frame", { Size = UDim2.new(1, 0, 0, 60), BackgroundColor3 = CONFIG.Colors.Surface,
        BackgroundTransparency = 0.2, ZIndex = 3, Parent = parent })
    Corner(frame, 8)
    Create("TextLabel", { Size = UDim2.new(1, -60, 0, 20), Position = UDim2.new(0, 12, 0, 6),
        BackgroundTransparency = 1, Text = text, TextColor3 = CONFIG.Colors.Text,
        Font = CONFIG.Fonts.Body, TextSize = IS_MOBILE and 13 or 14,
        TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 4, Parent = frame })
    local valueLabel = Create("TextLabel", { Size = UDim2.new(0, 50, 0, 20), Position = UDim2.new(1, -62, 0, 6),
        BackgroundTransparency = 1, Text = tostring(default), TextColor3 = CONFIG.Colors.Primary,
        Font = CONFIG.Fonts.Bold, TextSize = 14, TextXAlignment = Enum.TextXAlignment.Right, ZIndex = 4, Parent = frame })
    local barBg = Create("Frame", { Size = UDim2.new(1, -24, 0, 8), Position = UDim2.new(0, 12, 0, 38),
        BackgroundColor3 = CONFIG.Colors.SurfaceLight, ZIndex = 4, Parent = frame })
    Corner(barBg, 4)
    local barFill = Create("Frame", { Size = UDim2.new((default-min)/(max-min), 0, 1, 0),
        BackgroundColor3 = CONFIG.Colors.Primary, ZIndex = 5, Parent = barBg })
    Corner(barFill, 4)
    local knob = Create("Frame", { Size = UDim2.new(0, 14, 0, 14),
        Position = UDim2.new((default-min)/(max-min), -7, 0.5, -7),
        BackgroundColor3 = CONFIG.Colors.Text, ZIndex = 6, Parent = barBg })
    Corner(knob, 7)
    local dragging = false
    local function setValue(p)
        p = math.clamp(p, 0, 1)
        local value = math.floor(min + (max-min) * p + 0.5)
        barFill.Size = UDim2.new(p, 0, 1, 0)
        knob.Position = UDim2.new(p, -7, 0.5, -7)
        valueLabel.Text = tostring(value)
        if callback then callback(value) end
    end
    local inputArea = Create("TextButton", { Size = UDim2.new(1, 0, 0, 20), Position = UDim2.new(0, 0, 0, 28),
        BackgroundTransparency = 1, Text = "", ZIndex = 6, Parent = frame })
    inputArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then dragging = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local relX = input.Position.X - barBg.AbsolutePosition.X
            setValue(relX / barBg.AbsoluteSize.X)
        end
    end)
    setValue((default-min)/(max-min))
    return frame
end

local function ToggleFirstPerson(state)
    State.firstPersonEnabled = state
    if state then
        State.firstPersonSaved = LocalPlayer.CameraMaxZoomDistance
        LocalPlayer.CameraMinZoomDistance = 0.5
        LocalPlayer.CameraMaxZoomDistance = 0.5
        local function ApplyFPS()
            local char = LocalPlayer.Character
            if not char then return end
            local head = char:FindFirstChild("Head")
            if head and head:IsA("BasePart") then head.LocalTransparencyModifier = 1 end
            for _, acc in ipairs(char:GetChildren()) do
                if acc:IsA("Accessory") then
                    local handle = acc:FindFirstChild("Handle")
                    if handle and handle:IsA("BasePart") then handle.LocalTransparencyModifier = 1 end
                end
            end
        end
        ApplyFPS()
        Workspace.CurrentCamera.FieldOfView = State.firstPersonFOV
        if State.firstPersonConnection then State.firstPersonConnection:Disconnect() end
        State.firstPersonConnection = RunService.RenderStepped:Connect(function()
            if not State.firstPersonEnabled then return end
            ApplyFPS()
            if Workspace.CurrentCamera.FieldOfView ~= State.firstPersonFOV then
                Workspace.CurrentCamera.FieldOfView = State.firstPersonFOV
            end
        end)
    else
        if State.firstPersonConnection then State.firstPersonConnection:Disconnect(); State.firstPersonConnection = nil end
        LocalPlayer.CameraMaxZoomDistance = State.firstPersonSaved or 128
        LocalPlayer.CameraMinZoomDistance = 0.5
        Workspace.CurrentCamera.FieldOfView = 70
        local char = LocalPlayer.Character
        if char then
            local head = char:FindFirstChild("Head")
            if head and head:IsA("BasePart") then head.LocalTransparencyModifier = 0 end
            for _, acc in ipairs(char:GetChildren()) do
                if acc:IsA("Accessory") then
                    local handle = acc:FindFirstChild("Handle")
                    if handle and handle:IsA("BasePart") then handle.LocalTransparencyModifier = 0 end
                end
            end
        end
    end
end

local function ClearImmersive()
    if State.immersiveSaved then
        for k, v in pairs(State.immersiveSaved) do pcall(function() Lighting[k] = v end) end
        State.immersiveSaved = nil
    end
    for _, obj in pairs(State.immersiveObjects) do if obj and obj.Parent then obj:Destroy() end end
    State.immersiveObjects = {}
    if State.immersiveVignetteGui then State.immersiveVignetteGui:Destroy(); State.immersiveVignetteGui = nil; State.immersiveVignetteImage = nil end
end

local function ToggleFlashlight(state)
    State.flashlightEnabled = state
    if state then
        if not State.flashlightPart or not State.flashlightPart.Parent then
            local part = Instance.new("Part"); part.Name = "laxx_FlashlightSource"
            part.Anchored = true; part.CanCollide = false; part.CanTouch = false
            part.CanQuery = false; part.Transparency = 1; part.Size = Vector3.new(0.2, 0.2, 0.2)
            part.Parent = Workspace; State.flashlightPart = part
        end
        if not State.flashlightSpot or not State.flashlightSpot.Parent then
            local spot = Instance.new("SpotLight"); spot.Name = "laxx_FlashlightSpot"
            spot.Brightness = State.flashlightIntensity; spot.Range = State.flashlightRange
            spot.Angle = State.flashlightAngle; spot.Face = Enum.NormalId.Front
            spot.Shadows = true; spot.Parent = State.flashlightPart; State.flashlightSpot = spot
        end
        if State.flashlightConnection then State.flashlightConnection:Disconnect() end
        State.flashlightConnection = RunService.RenderStepped:Connect(function()
            if not State.flashlightEnabled then return end
            if not State.flashlightPart or not State.flashlightPart.Parent then return end
            State.flashlightPart.CFrame = Workspace.CurrentCamera.CFrame * CFrame.new(0, 0, -2)
        end)
    else
        if State.flashlightConnection then State.flashlightConnection:Disconnect(); State.flashlightConnection = nil end
        if State.flashlightSpot then State.flashlightSpot:Destroy(); State.flashlightSpot = nil end
        if State.flashlightPart then State.flashlightPart:Destroy(); State.flashlightPart = nil end
    end
end
local function UpdateFlashlightProperties()
    if State.flashlightSpot and State.flashlightSpot.Parent then
        State.flashlightSpot.Brightness = State.flashlightIntensity
        State.flashlightSpot.Range = State.flashlightRange
        State.flashlightSpot.Angle = State.flashlightAngle
    end
end

local function ToggleImmersionMode(state)
    State.immersionModeEnabled = state
    if state then
        State.immersionSavedTime = Lighting.ClockTime
        if not State.firstPersonEnabled then ToggleFirstPerson(true); State.immersionEnabledFirstPerson = true
        else State.immersionEnabledFirstPerson = false end
        pcall(function() Lighting.ClockTime = 0 end)
        if not State.flashlightEnabled then ToggleFlashlight(true); State.immersionEnabledFlashlight = true
        else State.immersionEnabledFlashlight = false end
    else
        if State.immersionEnabledFlashlight then ToggleFlashlight(false); State.immersionEnabledFlashlight = false end
        if State.immersionEnabledFirstPerson then ToggleFirstPerson(false); State.immersionEnabledFirstPerson = false end
        if State.immersionSavedTime then pcall(function() Lighting.ClockTime = State.immersionSavedTime end); State.immersionSavedTime = nil end
    end
end

local function ApplySkyPreset(presetName)
    local preset = SKY_PRESETS[presetName]
    if not preset then return end
    if not State.skySaved then
        local oldSky = Lighting:FindFirstChildOfClass("Sky")
        if oldSky then State.skySaved = oldSky:Clone(); oldSky:Destroy() end
    end
    if not State.skyObject or not State.skyObject.Parent then
        State.skyObject = Instance.new("Sky"); State.skyObject.Name = "laxx_Sky"; State.skyObject.Parent = Lighting
    end
    for prop, val in pairs(preset) do pcall(function() State.skyObject[prop] = val end) end
    pcall(function() State.skyObject.CelestialBodiesShown = false end)
end
local function RemoveSky()
    if State.skyObject then State.skyObject:Destroy(); State.skyObject = nil end
    if State.skySaved then State.skySaved.Parent = Lighting; State.skySaved = nil end
end
local function ToggleSky(state)
    State.skyEnabled = state
    if state then ApplySkyPreset(State.skyPreset) else RemoveSky() end
end
local function ChangeSkyPreset(name)
    State.skyPreset = name
    if State.skyEnabled then ApplySkyPreset(name) end
end

    end

local function GetClosestPlayerToCursor(fov)
    local cam = Workspace.CurrentCamera
    local mousePos = UserInputService:GetMouseLocation()
    local closest, shortestDist = nil, fov
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local skip = false
            if State.teamCheckEnabled and LocalPlayer.Team and player.Team == LocalPlayer.Team then skip = true end
            if not skip then
                local part = player.Character:FindFirstChild(State.aimbotPart) or player.Character:FindFirstChild("HumanoidRootPart")
                if part then
                    local canTarget = true
                    if State.wallCheckEnabled then
                        local params = RaycastParams.new()
                        params.FilterType = Enum.RaycastFilterType.Exclude
                        params.FilterDescendantsInstances = {LocalPlayer.Character}
                        local result = Workspace:Raycast(cam.CFrame.Position, part.Position - cam.CFrame.Position, params)
                        if result and result.Instance and not result.Instance:IsDescendantOf(player.Character) then
                            canTarget = false
                        end
                    end
                    if canTarget then
                        local screenPos, onScreen = cam:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                            if dist < shortestDist then shortestDist = dist; closest = part end
                        end
                    end
                end
            end
        end
    end
    return closest
end
local function ToggleAimbot(state)
    State.aimbotEnabled = state
    if state then
        if State.aimbotConnection then State.aimbotConnection:Disconnect() end
        State.aimbotConnection = RunService.RenderStepped:Connect(function()
            if not State.aimbotEnabled then return end
            local target = GetClosestPlayerToCursor(State.aimbotFOV)
            if target then
                local cam = Workspace.CurrentCamera
                local aimPos = cam.CFrame.Position
                local targetPos = target.Position
                if State.aimbotPrediction and target.Parent then
                    local hum = target.Parent:FindFirstChildOfClass("Humanoid")
                    if hum then targetPos = targetPos + hum.MoveDirection * State.aimbotPredictionValue * 10 end
                end
                local direction = (targetPos - aimPos).Unit
                cam.CFrame = cam.CFrame:Lerp(CFrame.new(aimPos, aimPos + direction), State.aimbotSmoothness)
                if State.autoShootEnabled then
                    pcall(function() VirtualUser:CaptureController(); VirtualUser:ClickButton1(Vector2.new()) end)
                end
            end
        end)
    else
        if State.aimbotConnection then State.aimbotConnection:Disconnect() end
    end
end

local function RemoveESPForPlayer(player)
    local obj = State.espObjects[player]
    if not obj then return end
    if obj.highlight then obj.highlight:Destroy() end
    if obj.billboard then obj.billboard:Destroy() end
    if obj.tracer then obj.tracer:Destroy() end
    State.espObjects[player] = nil
end
local function UpdateESPLoop()
    if not next(State.espObjects) then return end
    local myChar = LocalPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    for player, obj in pairs(State.espObjects) do
        if player.Character then
            local char = player.Character
            local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
            if hrp then
                if obj.distLbl and myHrp then
                    obj.distLbl.Text = math.floor((hrp.Position - myHrp.Position).Magnitude) .. "m"
                end
                if obj.tracer then
                    local screenPos, onScreen = Workspace.CurrentCamera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        obj.tracer.Visible = true
                        local startPos = Vector2.new(viewport.X/2, viewport.Y)
                        local endPos = Vector2.new(screenPos.X, screenPos.Y)
                        local delta = endPos - startPos
                        obj.tracer.Size = UDim2.new(0, delta.Magnitude, 0, 1)
                        obj.tracer.Position = UDim2.new(0, startPos.X, 0, startPos.Y)
                        obj.tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
                    else
                        obj.tracer.Visible = false
                    end
                end
            end
        end
    end
end
RunService.RenderStepped:Connect(UpdateESPLoop)
local function RefreshAllESP()
    for player, _ in pairs(State.espObjects) do RemoveESPForPlayer(player) end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then CreateESPForPlayer(player) end
    end
end

local function ExecuteCode(code)
    code = tostring(code or "")
    if code:gsub("%s", "") == "" then Notifier:Show("Executor", "Vide.", 2); return end
    if type(loadstring) ~= "function" then Notifier:Show("Executor", "loadstring indisponible.", 4); return end
    local func = loadstring(code)
    if not func then Notifier:Show("Executor", "Syntaxe invalide", 3); return end
    task.spawn(function()
        local success = pcall(func)
        if success then Notifier:Show("Executor", "OK !", 2)
        else Notifier:Show("Executor", "Erreur.", 3) end
    end)
end

                if not State.rivalsAntiKatana then return end

                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum then pcall(function() hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false) end) end
                end
            end)
        else
            if State.rivalsAntiKatanaConn then State.rivalsAntiKatanaConn:Disconnect(); State.rivalsAntiKatanaConn = nil end
        end
    end

        if state then
            if State.rivalsViewAngleConn then State.rivalsViewAngleConn:Disconnect() end
            State.rivalsViewAngleConn = RunService.RenderStepped:Connect(function()
                if not State.rivalsViewAngleSpoof then return end

            end)
        else
            if State.rivalsViewAngleConn then State.rivalsViewAngleConn:Disconnect(); State.rivalsViewAngleConn = nil end
        end

        local n = tonumber(text)
        if n then State.rivalsFpsValue = n; if State.rivalsFpsSpoof then ToggleRivalsFpsSpoof(true) end end

        local n = tonumber(text)
        if n then State.rivalsMsValue = n; if State.rivalsMsSpoof then ToggleRivalsMsSpoof(true) end end

        State.rivalsSkinEnabled = v

    Create("UIGridLayout", { SortOrder = Enum.SortOrder.LayoutOrder,
        CellSize = UDim2.new(0.32, -4, 0, 30), CellPadding = UDim2.new(0, 4, 0, 4), Parent = weaponFrame })

                b.BackgroundColor3 = CONFIG.Colors.SurfaceLight
                b.TextColor3 = CONFIG.Colors.TextDim

            wbtn.BackgroundColor3 = CONFIG.Colors.Rivals
            wbtn.TextColor3 = CONFIG.Colors.Text
        end)
    end
    Create("TextLabel", { Size = UDim2.new(1, 0, 0, 16), BackgroundTransparency = 1,
        Text = "Skin :", TextColor3 = CONFIG.Colors.Text, Font = CONFIG.Fonts.Bold,
        TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 3, Parent = rivalsContent })
    local skins = {"Default", "Gold", "Diamond", "Crimson", "Neon", "Galaxy", "Void", "Rainbow"}
    local skinFrame = Create("Frame", { Size = UDim2.new(1, 0, 0, 100),
        BackgroundColor3 = CONFIG.Colors.Surface, BackgroundTransparency = 0.2, ZIndex = 3, Parent = rivalsContent })
    Corner(skinFrame, 8)
    Create("UIGridLayout", { SortOrder = Enum.SortOrder.LayoutOrder,
        CellSize = UDim2.new(0.24, -4, 0, 30), CellPadding = UDim2.new(0, 4, 0, 4), Parent = skinFrame })
    Padding(skinFrame, 6, 6, 6, 6)

    for i, skin in ipairs(skins) do
        local sbtn = Create("TextButton", { BackgroundColor3 = CONFIG.Colors.SurfaceLight,
            Text = skin, TextColor3 = CONFIG.Colors.TextDim, Font = CONFIG.Fonts.Bold,
            TextSize = IS_MOBILE and 9 or 11, AutoButtonColor = false, LayoutOrder = i,
            ZIndex = 4, Parent = skinFrame })
        Corner(sbtn, 6)
        skinBtns[skin] = sbtn
        sbtn.MouseButton1Click:Connect(function()

                b.TextColor3 = CONFIG.Colors.TextDim
            end
            sbtn.BackgroundColor3 = CONFIG.Colors.Rivals
            sbtn.TextColor3 = CONFIG.Colors.Text
        end)
    end
    CreateTextBox(rivalsContent, "ID Skin personnalisé (optionnel)", "Ex: 123456789", "", function(text)
        State.rivalsCustomSkinId = tonumber(text) or text

        ApplyRivalsSkin()

    CreateSectionLabel(rivalsContent, "Wrap Changer", CONFIG.Colors.Rivals)
    CreateToggle(rivalsContent, "Enable Wrap Changer", State.rivalsWrapEnabled, function(v)
        State.rivalsWrapEnabled = v
        if v then ApplyRivalsWrap() end

    local colors = {

        MainFrame.Visible = true; MainFrame.Size = UDim2.new(0, 0, 0, 0)

end)
